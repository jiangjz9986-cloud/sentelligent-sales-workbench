## 1. 迁移 0030：users 表 + env 种子 + assignee 显示名回填

### 1.1 `backend/src/db/migrations/0030_users_table.mjs` 全文

```js
// v0.9.1 L1 认证层：users 表 + env 种子（首个 admin）+ action_items.assignee 显示名回填。
// 种子读 process.env（生产=cutover 后 backend 首启应用迁移，EnvironmentFile 在位）；
// env 缺失/非法（如 /dev/shm 彩排的 env-less 语境）时跳过种子，由 server.js 启动期
// ensureBootstrapAdmin 兜底补种——两处校验语义必须保持一致，改其一必改另一。
const ACCOUNT_RE = /^[a-z0-9]{2,32}$/;
const SCRYPT_RE = /^scrypt\$16384\$8\$1\$[A-Za-z0-9_-]{22}\$[A-Za-z0-9_-]{86}$/;

export function apply(db) {
  db.exec(`
    CREATE TABLE users (
      account TEXT PRIMARY KEY NOT NULL
        CHECK (length(account) BETWEEN 2 AND 32 AND account NOT GLOB '*[^a-z0-9]*'),
      display_name TEXT NOT NULL CHECK (length(display_name) BETWEEN 1 AND 50),
      password_hash TEXT NOT NULL CHECK (password_hash GLOB 'scrypt$16384$8$1$*'),
      role TEXT NOT NULL DEFAULT 'member' CHECK (role IN ('admin', 'member')),
      status TEXT NOT NULL DEFAULT 'active' CHECK (status IN ('active', 'disabled')),
      version INTEGER NOT NULL DEFAULT 1 CHECK (version >= 1),
      created_at TEXT NOT NULL,
      updated_at TEXT NOT NULL,
      last_login_at TEXT
    );
  `);
  const account = String(process.env.AUTH_ACCOUNT ?? "").trim();
  const envHash = String(process.env.AUTH_PASSWORD_HASH ?? "").trim();
  if (ACCOUNT_RE.test(account) && SCRYPT_RE.test(envHash)) {
    const now = new Date().toISOString();
    db.prepare(`
      INSERT INTO users (account, display_name, password_hash, role, status, created_at, updated_at)
      VALUES ($account, '继振', $envHash, 'admin', 'active', $now, $now)
    `).run({ $account: account, $envHash: envHash, $now: now });
  }
  // 0029 已把 owner/assignee 统一为账号 id；有 display_name 后把展示列换回人名。
  // 子查询以 assignee∈users.account 为闸：种子缺席时空转、二跑幂等（'继振'∉account）。
  db.exec(`
    UPDATE action_items
       SET assignee = (SELECT display_name FROM users WHERE users.account = action_items.assignee)
     WHERE assignee IN (SELECT account FROM users)
  `);
}
```

要点：账号规范=小写字母数字 2–32（现 `AUTH_ACCOUNT=jiangjz` 天然合规；不做大小写转换，非法即跳过种子并靠 §2 兜底）；scrypt 正则=`password.js` 规范编码（salt 16B→22 字符、key 64B→86 字符 base64url）；不建额外索引（表≤十行量级）；哈希直接搬运、算法不变。**闭环 audit A #11**：`upsertActionFromQuickRecord` 的运行时改动见 §3.4。

## 2. `backend/src/auth/usersStore.js`（新模块）与启动兜底种子

同层放置（auth/ 已有 session/password/loginRateLimit），全同步、db 首参，SQL 单文件收敛：

| 函数 | 契约 |
|---|---|
| `getUser(db, account)` | 精确匹配返回整行（含 password_hash，**仅内部使用，HTTP 层永不序列化**）或 null |
| `listUsers(db)` | 全表按 account 升序，**不含 password_hash**，返回 camelCase 行 |
| `createUser(db, {account, displayName, passwordHash, role, now})` | INSERT；PK 冲突抛 SQLITE_CONSTRAINT 由调用方映射 409 |
| `updateUserVersioned(db, {account, expectedVersion, set, now})` | 自带乐观锁（users PK 是 account 非 id，**不复用** server.js `runVersionedUpdate`）：`UPDATE … SET …, version=version+1, updated_at=$now WHERE account=$account AND version=$expectedVersion`；changes=0 时回读区分——无行→NotFound、有行→VersionConflict(currentVersion)，错误类型与 `itineraryRepositoryFailure` 映射惯例对齐 |
| `recordLastLogin(db, account, nowIso)` | `UPDATE users SET last_login_at=$now WHERE account=$account`（不 bump version，非业务编辑） |
| `countActiveAdmins(db)` | `role='admin' AND status='active'` 计数 |
| `ensureBootstrapAdmin(db, config, {now})` | 启动兜底：config.authAccount 合规（ACCOUNT_RE）且 `validatePasswordHashEncoding(config.authPasswordHash)` 通过且 `getUser` 无行 → `createUser`(display_name='继振', role=admin) + `insertAudit(user.create, actor="system:bootstrap")` + console.warn；否则 no-op。**只插不改**——绝不覆盖已有行的密码（否则 UI 改密会被每次重启还原） |

装配：`server.js` `createServer` 内 `openDatabase` 之后立即 `ensureBootstrapAdmin(db, config)`。三重种子保障=0030 env 种子 → 每次启动兜底 → 手工 SQL（§8 回滚节附全文）。会话吊销新增 `auth/session.js` `revokeSessionsForAccount(db, account, { exceptSessionId, now })`（按 account 列直查，无需 token 哈希故不收 config）：`UPDATE auth_sessions SET revoked_at=$now WHERE account=$account AND revoked_at IS NULL [AND id != $exceptSessionId]`，返回 changes。

## 3. 登录改造（server.js）

### 3.1 `authenticateLogin` 新流程（限流四步原封不动）

```
账号 trim / limiterKeys 两枚 / pruneLoginRateLimits / assertLoginAllowed   ← 现状照抄
const user = getUser(db, account);
let fallback = false, valid = false;
if (user) {
  const match = await verifyPassword(body?.password, user.passwordHash);   // 先算再判 status，保持恒时
  valid = match && user.status === "active";
} else {
  valid = await configuredCredentialsMatch(config, body);                  // 原函数原样保留 = env 回退轨
  fallback = valid;
}
失败：recordLoginFailure ×2 → return null（响应仍统一 401 INVALID_CREDENTIALS，停用/未知/密码错不可区分，防枚举）
成功：clearLoginFailures ×2；user 存在则 recordLastLogin；
fallback 命中 → console.warn("category=auth env fallback login …") + insertAudit(auth.login.env_fallback)
return { session: createSession({account: user?.account ?? config.authAccount}), displayName, role, fallback }
```

恒时论证：三条失败路径（行存在密码错 / 行存在但停用 / 无行走 env 比对）都恰好执行一次 scrypt `verifyPassword`，与现状等时；`createSession`/`getActiveSession`/CSRF/7 天 TTL 零改动，**现有生产会话在升级后继续有效**（account 值不变）。

### 3.2 响应契约（登录与会话两端点同步加 role）

`POST /api/auth/login` 与 `GET /api/auth/session` 响应统一为 `{ account, displayName, role, expiresAt, csrfToken }`：displayName=`users.display_name ?? account`，role=`users.role ?? "member"`（session 端点每请求查一次 users，表极小；角色变更**即时生效免重登**）。回退轨会话无 users 行 → displayName=account、role=member（管理能力靠重启兜底种子恢复，见 §8）。`requestSchemas.login` **不动**。全仓检查前后端对 `displayName` 的既有断言并更新（登录响应从账号名变为"继振"）。

### 3.3 生产 config 校验取舍（裁定：本版保留强制）

`validateProductionConfig` 与 `hasCookieAuthConfiguration`/`isAuthEnabled` **一字不改**：AUTH_ACCOUNT/AUTH_PASSWORD_HASH 本版仍是生产强制项。理由：(a) 它们是 0030 种子与启动兜底的数据源；(b) 双轨回退需要它们兜锁死风险；(c) 503 AUTH_NOT_CONFIGURED 语义不漂移，部署零配置变更。**退役路径**（写入 release 文档遗留项）：v0.9.3 移除回退轨时同步改造——`isAuthEnabled` 改为"users 有 active 行 或 env 在位"，AUTH_ACCOUNT 降级为可选 bootstrap 键，启动期加"users 非空"运行时检查；届时再评估，不在本版做。

### 3.4 深写回 assignee 终态（v0.9.0 §2.4 遗留归本版）

`upsertActionFromQuickRecord` 的 `$assignee`（v0.9.0 已改为 `quickRecord.owner`）本版改为：`getUser(db, quickRecord.owner)?.displayName ?? quickRecord.owner ?? null`。存量回填已由 0030 完成。`risk_items.assignee`（种子展示名"继振"×3）不动。

## 4. admin API 与角色门禁

### 4.1 `requireAdminRole(db, request)`（server.js 内新助手）

`request.authContext.kind !== "user"` → 401（机器令牌本就被路由白名单挡在外——admin 路由**不加入**任何 `INTEGRATION_ROUTES`，机器打之 403 MACHINE_SCOPE_DENIED，现机制自动生效）；`getUser(db, authContext.account)` 的 `role !== 'admin' || status !== 'active'` → 403 `ADMIN_ROLE_REQUIRED`。角色每请求从表读取，无缓存。

### 4.2 端点契约（挂 server.js if-链 auth/logout 分支之后，风格与 settings 域一致）

| 端点 | 鉴权 | 行为 |
|---|---|---|
| `GET /api/admin/users` | admin | 200 `{ items: [{account, displayName, role, status, createdAt, updatedAt, lastLoginAt, version}] }`，account 升序 |
| `POST /api/admin/users` | admin | 校验 `adminUserCreate` + 语义（account 匹配 `^[a-z0-9]{2,32}$`、密码策略 10–128 字符）→ `hashPassword`（**事务外，scrypt 异步**）→ `withImmediateTransaction`：createUser + `insertAudit(user.create)` → 201 `{ item }`（无 password 字段）；PK 冲突 → 409 `USER_EXISTS`。不接 Idempotency-Key（PK 即幂等闸，重复提交=409） |
| `PATCH /api/admin/users/:account` | admin | 校验 `adminUserPatch`（expectedVersion 必填 + 至少一个可变字段）；password 在场则事务外先 hash；`withImmediateTransaction` 内按序：①`getUser(target)` 无行→404 `USER_NOT_FOUND`；②防呆二则（§4.3）；③`updateUserVersioned`（冲突→409 `VERSION_CONFLICT`+currentVersion）；④status→disabled 或 password 在场 → `revokeSessionsForAccount(target)`（全量吊销；**例外**：target===操作者本人重置自己密码时 `exceptSessionId=authContext.id` 保留当前会话，防 admin 自我重置后被踢出中断操作）；⑤审计（§5 映射）→ 200 `{ item }` |
| `POST /api/auth/change-password` | 任意登录用户 | 校验 `changePassword`；入口先 `assertLoginAllowed`（复用登录限流键 account+address，防拿到会话后暴破旧密码）；`getUser(自己)` 无行（回退轨会话）→ 409 `USER_NOT_PROVISIONED`；`verifyPassword(currentPassword)` 失败 → `recordLoginFailure`×2 + 403 `CURRENT_PASSWORD_INCORRECT`（**绝不可用 401**：`salesWorkbenchApi.requestApi` 对一切 401 调 `invalidateSession` 全局登出，旧密码敲错会被踢出）；成功 → 事务外 hash 新密码 → 事务内 updateUserVersioned + `revokeSessionsForAccount(自己, {exceptSessionId: authContext.id})` + 审计 → 200 `{ ok: true, revokedSessions }`。**会话保留裁定：改密后吊销本人其他会话、当前会话保留**（被盗会话改密自锁 + 改密者不被打断，推荐项采纳） |

### 4.3 防呆规则（事务内强一致）

1. **不能停用自己**：target===authContext.account 且 patch.status='disabled' → 409 `SELF_DISABLE_FORBIDDEN`（无论 admin 有几个）。
2. **不能停用/降级最后一个 active admin**：target 现为 active admin，且 patch 将其 status→disabled 或 role→member，且 `countActiveAdmins(db)===1` → 409 `LAST_ADMIN_PROTECTED`。自降级在他 admin 存在时允许（即时失去管理面，文档标注）。admin 可重置自己密码（等价改密免旧密码，权限自洽）。

### 4.4 `validation/requests.js` 新 schema（沿 text()/enum 惯例）

```js
adminUserCreate: {
  account: text(32, { required: true }), displayName: text(50, { required: true }),
  password: text(128, { required: true }),
  role: { type: "enum", values: ["admin", "member"] },          // 缺省 member
},
adminUserPatch: {
  displayName: text(50, { nonEmpty: true }), role: { type: "enum", values: ["admin", "member"] },
  status: { type: "enum", values: ["active", "disabled"] }, password: text(128, { nonEmpty: true }),
  expectedVersion: { type: "integer", min: 1, required: true },
},
changePassword: { currentPassword: text(1000, { required: true }), newPassword: text(128, { required: true }) },
```

语义追加（schema 表达不了的，端点内 `validationFailure` 惯例）：account 正则、密码最短 10 字符（fields 对该键报 policy 违规）、adminUserPatch 除 expectedVersion 外至少一键。

### 4.5 系统配置域写端点门禁（决策点，裁定：做）

member 若能改 DeepSeek Key/PushPlus/微信绑定/招标调度，等于系统级权限，UI 藏而后端不拦=纸闸。本版给下列**写端点**统一加 `requireAdminRole`（读端点不动，随 v0.9.2 隔离一并收口）：`PUT|POST|DELETE /api/settings/deepseek-key(+别名)`、`PUT|POST|DELETE /api/settings/pushplus-token(+别名)`、`POST /api/settings/pushplus/test`、`GET|POST|DELETE /api/integrations/weixin-agent/login`（绑定生命周期整组，GET 含启动态语义故一并管）、`PATCH /api/hospital-tenders/scheduler`、`POST /api/hospital-tenders/scheduler/run(-next)`、`POST /api/hospital-tenders/run`。生产现状单账号=admin，行为零变化。

### 4.6 错误码词表（新增部分；既有 401/422/429/409 VERSION_CONFLICT 惯例不变）

| 码 | HTTP | 场景 |
|---|---|---|
| ADMIN_ROLE_REQUIRED | 403 | 非 active admin 调 §4.2/§4.5 门禁端点 |
| CURRENT_PASSWORD_INCORRECT | 403 | 改密旧密码错（403 而非 401，避免前端全局登出） |

| USER_NOT_PROVISIONED | 409 | 回退轨会话（无 users 行）调改密 |

## 5. 审计动作词表（全量入 audit_logs，entity_type="user"，entity_id=目标账号，actor=操作者账号）

`user.create`（after={account,displayName,role,status}；启动兜底种子同 action、actor="system:bootstrap"）、`user.update`（displayName/role 变更，before/after 限已变字段）、`user.disable` / `user.enable`（status 迁移专用）、`password.reset`（admin 重置他人，after={sessionsRevoked}，**哈希绝不入审计**——auditRepository 敏感键正则本就剔除 password 系键，勿把哈希放非 password 命名键里）、`password.change`（本人）、`auth.login.env_fallback`（metadata={reason:"user_row_missing"}）。组合 PATCH 允许一次产生多行（status+password 同改 → disable+reset 两行），语义精确优先。均在既有 `insertAudit` 惯例内，actor 词表符合 L0 规范（账号 id | system:*）。
